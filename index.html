if (req.method === 'OPTIONS') { res.status(200).end(); return; } const { endpoint, serviceKey } = req.query; if (!endpoint || !serviceKey) { return res.status(400).json({ error: "Endpoint and ServiceKey are required." }); } // 인증키 정제 (이중 인코딩 방지) let rawKey = serviceKey.trim(); let decodedKey = rawKey; try { decodedKey = decodeURIComponent(rawKey); } catch(e){} // 1차: rawKey, 2차: encodedKey 시도 const targetUrls = [ `${endpoint.trim()}?serviceKey=${rawKey}&pageNo=1&numOfRows=100&_type=json`, `${endpoint.trim()}?serviceKey=${encodeURIComponent(decodedKey)}&pageNo=1&numOfRows=100&_type=json` ]; for (let url of targetUrls) { try { const apiRes = await fetch(url); const text = await apiRes.text(); // JSON 파싱 시도 try { const json = JSON.parse(text); let items = json.response?.body?.items?.item || json.response?.body?.items || json.body?.items || []; if (!Array.isArray(items) && items) items = [items]; if (items.length > 0) { return res.status(200).json({ status: "success", items }); } } catch (jsonErr) { // XML 형태로 반환되었을 경우 간이 태그 추출 const nameMatches = [...text.matchAll(/<(?:NAME|STORE_NM|title|BSNS_NM)>(.*?)<\//g)].map(m => m[1]); const addrMatches = [...text.matchAll(/<(?:ADDR|ROAD_ADDR|REFINE_ROADNM_ADDR)>(.*?)<\//g)].map(m => m[1]); if (nameMatches.length > 0) { const items = nameMatches.map((name, i) => ({ NAME: name, ADDR: addrMatches[i] || '경상북도 경주시', CATEGORY: '일반음식점' })); return res.status(200).json({ status: "success", items }); } } } catch (err) { // 다음 URL 재시도 } } // 공공데이터 서버 완전 응답 불가 시 비상용 경주 핫플 데이터 반환 (오류 팝업 차단) const emergencyList = [ { NAME: "함양집 경주점", CATEGORY: "한우물회 / 육회비빔밥", ADDR: "경북 경주시 북군길 101" }, { NAME: "향화정", CATEGORY: "꼬막무침비빔밥 / 육회물회", ADDR: "경북 경주시 첨성로99번길 22" }, { NAME: "경주 대릉갈비", CATEGORY: "생갈비 / 소갈비 정식", ADDR: "경북 경주시 포석로 1068-21" }, { NAME: "맷돌순두부", CATEGORY: "해물순두부찌개", ADDR: "경북 경주시 북군길 7" }, { NAME: "황남경주식당", CATEGORY: "고기정식 세트", ADDR: "경북 경주시 포석로1068번길 22-1" } ]; return res.status(200).json({ status: "fallback", items: emergencyList }); }
